    Caso de uso

    Objetivo de la práctica

    Cargar CSV directamente en Elasticsearch desde Kibana y hacer todas las transformaciones dentro del stack Elastic hasta obtener un índice final listo para análisis:

    • Índices RAW (CSV “tal cual”):

    • covid_raw

    • prov_raw

    • trams_raw

    • Índices REF (limpios para enrich):

    • prov_ref
    • trams_ref
    • Índice final (curado):
    • covid_enriched

    El índice final covid_enriched debe incluir:

    • @timestamp (date)
    • provincia.* enriquecido (nombre, CCAA, ISO3166-2)
    • alarma.cod (integer) y alarma.desc
    • métricas num_* como integers
    • trazabilidad opcional: fecha_raw, ingest_error

    Importación de CSV y creación de índices RAW

    Importación con Data Visualizer

    Kibana: - Machine Learning → Data Visualizer → Upload file - Subir cada CSV y asignar:

    CSV Índice RAW
    Serie COVID covid_raw
    Provincias prov_raw
    Tramos alarma trams_raw

    Si Kibana muestra timeout, comprueba igualmente en Index Management: la importación puede haber terminado.

    Verificación

    GET _cat/indices/covid_raw,prov_raw,trams_raw?v

    Ver un documento de muestra

    GET covid_raw/_search?size=1
GET prov_raw/_search?size=1
GET trams_raw/_search?size=1

    Crear índices REF (limpios) para enrich

    ¿Por qué REF?

    • Evita depender de nombres de columnas con espacios/acentos (p.ej. “Código 3166-2”).
    • Tipos claros (keyword, integer) para enrich.
    • Proceso reproducible: RAW intactos + REF limpios.

    Construcción de prov_ref

    Crear índice

    DELETE prov_ref

PUT prov_ref
{
  "mappings": {
    "dynamic": "strict",
    "properties": {
      "provincia_code": { "type": "keyword" },
      "iso3166_2":      { "type": "keyword" },
      "nombre":         { "type": "keyword" },
      "ccaa":           { "type": "keyword" }
    }
  }
}

    Pipeline prov_ref_pipeline

    PUT _ingest/pipeline/prov_ref_pipeline
{
  "processors": [
    {
      "script": {
        "lang": "painless",
        "source": """
          def iso = ctx['Código 3166-2'];
          def nombre = ctx['Nombre de la subdivisión'];
          def ccaa = ctx['Subdivisión superior'];

          String isoStr = iso == null ? null : iso.toString();
          String code = null;

          if (isoStr != null) {
            if (isoStr.startsWith('ES-') && isoStr.length() > 3) code = isoStr.substring(3);
            else code = isoStr;
          }

          if (code != null) ctx.provincia_code = code;
          if (isoStr != null) ctx.iso3166_2 = isoStr;
          if (nombre != null) ctx.nombre = nombre.toString();
          if (ccaa != null) ctx.ccaa = ccaa.toString();
        """
      }
    },
    {
      "remove": {
        "field": [
          "Código 3166-2",
          "Categoría de la subdivisión",
          "Nombre de la subdivisión",
          "Código del idioma",
          "Subdivisión superior"
        ],
        "ignore_missing": true
      }
    }
  ]
}

    Reindex prov_raw → prov_ref

    POST _reindex
{
  "source": { "index": "prov_raw" },
  "dest":   { "index": "prov_ref", "pipeline": "prov_ref_pipeline" },
  "conflicts": "proceed"
}

    Validación

    GET prov_ref/_search?size=5

    Construcción de trams_ref

    Crear índice

    DELETE trams_ref

PUT trams_ref
{
  "mappings": {
    "dynamic": "strict",
    "properties": {
      "cod":  { "type": "integer" },
      "desc": { "type": "keyword" }
    }
  }
}

    Pipeline trams_ref_pipeline

    PUT _ingest/pipeline/trams_ref_pipeline
{
  "processors": [
    {
      "script": {
        "lang": "painless",
        "source": """
          if (ctx.containsKey('cod') && ctx.cod != null) {
            def c = ctx.cod;
            if (c instanceof Number) {
              ctx.cod = ((Number)c).intValue();
            } else {
              ctx.cod = Integer.parseInt(c.toString().trim());
            }
          }
          if (ctx.containsKey('desc') && ctx.desc != null) {
            ctx.desc = ctx.desc.toString();
          }
        """
      }
    }
  ]
}

    Reindex trams_raw → trams_ref

{
  "source": { "index": "trams_raw" },
  "dest":   { "index": "trams_ref", "pipeline": "trams_ref_pipeline" },
  "conflicts": "proceed"
}

    Crear el índice final covid_enriched

    DELETE covid_enriched

PUT covid_enriched
{
  "mappings": {
    "properties": {
      "@timestamp": { "type": "date" },

      "provincia": {
        "properties": {
          "code":      { "type": "keyword" },
          "iso3166_2": { "type": "keyword" },
          "nombre":    { "type": "keyword" },
          "ccaa":      { "type": "keyword" }
        }
      },

      "sexo": { "type": "keyword" },
      "grupo_edad": { "type": "keyword" },
      "grupo_edad_orden": { "type": "integer" },

      "alarma": {
        "properties": {
          "cod":  { "type": "integer" },
          "desc": { "type": "keyword" }
        }
      },

      "num_casos": { "type": "integer" },
      "num_hosp":  { "type": "integer" },
      "num_uci":   { "type": "integer" },
      "num_def":   { "type": "integer" },

      "fecha_raw":   { "type": "keyword" },
      "ingest_error":{ "type": "keyword" }
    }
  }
}

    Pipeline final covid_pipeline

    Creación

    DELETE _ingest/pipeline/covid_pipeline

PUT _ingest/pipeline/covid_pipeline
{
  "description": "COVID raw -> enriched (timestamp + enrich provincias + tramo alarma + tipos int)",
  "processors": [
    {
      "script": {
        "lang": "painless",
        "source": """
          // Guarda la fecha original por trazabilidad
          if (ctx.containsKey('fecha') && ctx.fecha != null) {
            ctx.fecha_raw = ctx.fecha.toString();
            ctx.fecha = ctx.fecha.toString().trim();
          }

          // Normaliza provincia
          if (ctx.containsKey('provincia_iso') && ctx.provincia_iso != null) {
            ctx.provincia_iso = ctx.provincia_iso.toString().trim().toUpperCase();
          }

          // Métricas -> int (acepta Number, string, string vacío)
          def nums = ['num_casos','num_hosp','num_uci','num_def'];
          for (def f : nums) {
            if (!ctx.containsKey(f) || ctx[f] == null) continue;
            def v = ctx[f];
            try {
              if (v instanceof Number) ctx[f] = ((Number)v).intValue();
              else {
                def s = v.toString().trim();
                ctx[f] = (s.length() == 0) ? 0 : Integer.parseInt(s);
              }
            } catch (Exception e) {
              ctx[f] = 0;
              ctx.ingest_error = 'metric_parse_failed';
            }
          }

          // @timestamp desde fecha: primeros 10 chars y valida con LocalDate.parse()
          if (ctx.containsKey('fecha') && ctx.fecha != null) {
            String s = ctx.fecha.toString().trim();
            String d = (s.length() >= 10) ? s.substring(0,10) : s;

            try {
              java.time.LocalDate.parse(d);
              ctx['@timestamp'] = d + 'T00:00:00.000Z';
            } catch (Exception e) {
              ctx.ingest_error = 'date_parse_failed';
            }
          } else {
            ctx.ingest_error = 'missing_fecha';
          }

          ctx.remove('fecha');
        """
      }
    },

    { "rename": { "field": "provincia_iso", "target_field": "provincia_code", "ignore_missing": true } },

    {
      "script": {
        "lang": "painless",
        "source": """
          if (ctx.containsKey('provincia_code') && ctx.provincia_code != null) {
            ctx.provincia_code = ctx.provincia_code.toString().trim().toUpperCase();
          }
        """
      }
    },

    { "enrich": { "policy_name": "prov_by_code", "field": "provincia_code", "target_field": "provincia" } },

    {
      "script": {
        "lang": "painless",
        "source": """
          if (ctx.provincia == null) ctx.provincia = new HashMap();
          if (ctx.containsKey('provincia_code') && ctx.provincia_code != null) {
            ctx.provincia.put('code', ctx.provincia_code.toString());
          }
          ctx.remove('provincia_code');

          // grupo_edad_orden
          if (ctx.grupo_edad != null) {
            def m = /^(\\d+)/.matcher(ctx.grupo_edad.toString());
            if (m.find()) ctx.grupo_edad_orden = Integer.parseInt(m.group(1));
          }

          // alarma_cod (ENTERO)
          int cod = 99;
          if (ctx.containsKey('@timestamp') && ctx['@timestamp'] != null) {
            String ts = ctx['@timestamp'].toString();
            String dp = ts.substring(0,10);
            def date = java.time.LocalDate.parse(dp);

            def d1  = java.time.LocalDate.parse('2020-03-14');
            def d2  = java.time.LocalDate.parse('2020-03-27');
            def d3  = java.time.LocalDate.parse('2020-04-10');
            def d4  = java.time.LocalDate.parse('2020-04-24');
            def d5  = java.time.LocalDate.parse('2020-05-08');
            def d6  = java.time.LocalDate.parse('2020-05-22');
            def d7  = java.time.LocalDate.parse('2020-06-05');
            def end = java.time.LocalDate.parse('2020-06-21');

            if (date.isBefore(d1) || date.isAfter(end)) cod = 99;
            else if (!date.isBefore(d7)) cod = 7;
            else if (!date.isBefore(d6)) cod = 6;
            else if (!date.isBefore(d5)) cod = 5;
            else if (!date.isBefore(d4)) cod = 4;
            else if (!date.isBefore(d3)) cod = 3;
            else if (!date.isBefore(d2)) cod = 2;
            else cod = 1;
          }
          ctx.alarma_cod = cod;
        """
      }
    },

    { "enrich": { "policy_name": "tram_by_cod", "field": "alarma_cod", "target_field": "alarma_meta" } },

    {
      "script": {
        "lang": "painless",
        "source": """
          if (ctx.alarma == null) ctx.alarma = new HashMap();

          int cod = (ctx.alarma_cod instanceof Number)
            ? ((Number)ctx.alarma_cod).intValue()
            : Integer.parseInt(ctx.alarma_cod.toString().trim());

          ctx.alarma.put('cod', cod);

          if (ctx.alarma_meta != null && ctx.alarma_meta.desc != null) {
            ctx.alarma.put('desc', ctx.alarma_meta.desc.toString());
          }

          ctx.remove('alarma_cod');
          ctx.remove('alarma_meta');
        """
      }
    }
  ]
}

    Monitorización

    GET _tasks?actions=*reindex&detailed=true
GET covid_raw/_count
GET covid_enriched/_count

    Criterio de éxito: covid_enriched/_count ≈ covid_raw/_count.

    Control de calidad del índice final

    Docs sin @timestamp

    GET covid_enriched/_count
{
  "query": { "bool": { "must_not": [ { "exists": { "field": "@timestamp" } } ] } }
}

    Docs con ingest_error

    GET covid_enriched/_count
{
  "query": { "exists": { "field": "ingest_error" } }
}

{
  "size": 0,
  "aggs": {
    "by_month": {
      "date_histogram": {
        "field": "@timestamp",
        "calendar_interval": "month",

  }

    Visualización en Kibana (Lens)

    Crear Data View

    Kibana: - Stack Management → Data Views → Create data view - Pattern: covid_enriched - Time field: @timestamp

    Gráficos recomendados

    1) Serie temporal de casos: - X: @timestamp - Y: Sum(num_casos)

    2) Top provincias: - Breakdown: Top values provincia.nombre (poner Top 60) - activar Other bucket y Missing bucket (evita huecos visuales)

    3) Por grupo de edad: - Breakdown: grupo_edad (ordenar con grupo_edad_orden)

    4) Por tramo alarma: - Breakdown: alarma.desc o alarma.cod
